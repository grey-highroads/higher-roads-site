  name: github
  repo: YOUR-GITHUB-USERNAME/steve-taylor-presents
  branch: main
  base_url: https://steve-taylor-presents.pages.dev
  auth_endpoint: /api/auth

media_folder: "images/uploads"
public_folder: "/images/uploads"

Important Notes

  • Use functions/api, not functions/API.
  • The Cloudflare OAuth routes are auth.js and callback.js.
  • Do not create an empty [...path].js file unless following a different OAuth implementation.
  • The GitHub Client ID is a text variable.
  • The GitHub Client Secret is a secret variable.
  • Client users need GitHub accounts with this setup.
